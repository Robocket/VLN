cmake_minimum_required(VERSION 3.8)
project(vln_planner)
find_package(ament_cmake REQUIRED)
find_package(rclcpp REQUIRED)
find_package(sensor_msgs REQUIRED)
find_package(geometry_msgs REQUIRED)
find_package(nav_msgs REQUIRED)
add_executable(a_star_planner_node src/a_star_planner.cpp)
target_compile_features(a_star_planner_node PUBLIC cxx_std_17)
target_compile_options(a_star_planner_node PRIVATE -Wall -Wextra -Wpedantic)
ament_target_dependencies(a_star_planner_node rclcpp sensor_msgs geometry_msgs nav_msgs)
install(TARGETS a_star_planner_node DESTINATION lib/${PROJECT_NAME})
ament_package()
